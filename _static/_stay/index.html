#!/usr/bin/env bash set -euo pipefail readonly INSTALL_DIR="/usr/local/bin" readonly TARGET="${INSTALL_DIR}/strapped" readonly SOURCE="https://raw.githubusercontent.com/azohra/strapped.sh/main/strapped" fail() { printf 'installer: %s\n' "$1" >&2 exit 1 } [ "$(uname -s)" = "Darwin" ] || fail "refusing — this installer supports macOS only" [ -d "${INSTALL_DIR}" ] || fail "refusing — ${INSTALL_DIR} does not exist; create it and retry" [ -w "${INSTALL_DIR}" ] || fail "refusing — ${INSTALL_DIR} is not writable; fix its ownership and retry" case ":${PATH}:" in *":${INSTALL_DIR}:"*) ;; *) fail "refusing — ${INSTALL_DIR} is not in PATH; add it and retry" ;; esac download=$(mktemp "${INSTALL_DIR}/.strapped.XXXXXX") || fail "could not create a temporary download" trap 'rm -f "${download}"' EXIT curl -fsSL "${SOURCE}" --output "${download}" || fail "could not download strapped" chmod 0755 "${download}" || fail "could not make the download executable" mv "${download}" "${TARGET}" || fail "could not install strapped" trap - EXIT printf 'Installed strapped at %s\n' "${TARGET}"
